image:       apache/airflow:2.7.3
container_name: airflow-webserver
command:     webserver
ports:       8081 on your laptop → 8080 inside container
             (8080 is taken by Spark, so use 8081 externally)
depends_on:  postgres, airflow-scheduler
environment:
  AIRFLOW__CORE__EXECUTOR: LocalExecutor
  AIRFLOW__DATABASE__SQL_ALCHEMY_CONN: postgresql+psycopg2://airflow:airflow@postgres/airflow
  AIRFLOW__CORE__LOAD_EXAMPLES: 'false'
volumes:
  - ./dags:/opt/airflow/dags       (bind mount — you write DAGs locally)
  - airflow_logs:/opt/airflow/logs (named volume — persist logs)
healthcheck:
  test command: curl -f http://localhost:8080/health
  interval: 30s, timeout: 10s, retries: 3
restart: unless-stopped

image:       apache/airflow:2.7.3
container_name: airflow-scheduler
command:     scheduler
depends_on:  postgres
same environment as webserver
same volumes as webserver
restart: unless-stopped
ports: NOT needed

depends_on:  postgres
same environment as webserver
